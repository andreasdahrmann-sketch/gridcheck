// ═══════════════════════════════════════════════════

let state = {
  project: {},
  netParams: {},
  result: null,
  calcId: null

let history = JSON.parse(localStorage.getItem('gc_history') || '[]');

// ═══════════════════════════════════════════════════
// NAVIGATION
// ═══════════════════════════════════════════════════
function goStep(n) {
  [1,2,3].forEach(i => {
    document.getElementById('panel'+i).classList.remove('active');
    document.getElementById('tab'+i).classList.remove('active','done');
  });
  document.getElementById('panel'+n).classList.add('active');
  document.getElementById('tab'+n).classList.add('active');
  for(let i=1;i<n;i++) document.getElementById('tab'+i).classList.add('done');
  window.scrollTo({top:0,behavior:'smooth'});
}

function newProject() {
  document.getElementById('f-name').value = '';
  document.getElementById('f-plz').value = '';
  document.getElementById('f-power').value = '';
  goStep(1);
}

// ═══════════════════════════════════════════════════
// STEP 1 VALIDATION
// ═══════════════════════════════════════════════════
function step1Next() {
  const name  = document.getElementById('f-name').value.trim();
  const plz   = document.getElementById('f-plz').value.trim();
  const power = parseFloat(document.getElementById('f-power').value);
  const cosphi= parseFloat(document.getElementById('f-cosphi').value);

  if (!name)                    { alert('❌ Bitte Projektname eingeben.'); return; }
  if (!/^\d{5}$/.test(plz))    { alert('❌ PLZ muss 5 Ziffern haben.'); return; }
  if (!power || power <= 0)     { alert('❌ Bitte gültige Anschlussleistung eingeben.'); return; }
  if (power > 500000)           { alert('❌ Leistung max. 500.000 kW'); return; }
  if (cosphi < 0.8 || cosphi > 1.0) { alert('❌ cos φ: 0,80 – 1,00'); return; }

  state.project = { name, plz, power, type: document.getElementById('f-type').value,
                    voltage: parseFloat(document.getElementById('f-voltage').value), cosphi };
  goStep(2);
}

// ═══════════════════════════════════════════════════
// BERECHNUNG (vollständig lokal, keine API)
// ═══════════════════════════════════════════════════
function runCalc() {
  // Eingaben lesen
  const sk_vap   = parseFloat(document.getElementById('f-sk').value);
  const lengthKm = parseFloat(document.getElementById('f-length').value);
  const trafoSn  = parseFloat(document.getElementById('f-trafo').value);   // kVA
  const uk       = parseFloat(document.getElementById('f-uk').value) / 100;
  const existing = parseFloat(document.getElementById('f-existing').value) || 0;
  const rx       = parseFloat(document.getElementById('f-rx').value);
  const parallel = parseInt(document.getElementById('f-parallel').value) || 1;

  const cableEl = document.getElementById('f-cable');
  const selOpt  = cableEl.options[cableEl.selectedIndex];
  const r_km    = parseFloat(selOpt.dataset.r);   // Ω/km
  const x_km    = parseFloat(selOpt.dataset.x);   // Ω/km
  const i_max_1 = parseFloat(selOpt.dataset.imax); // A je Strang
  const i_max   = i_max_1 * parallel;

  const { power, voltage, cosphi, name, plz, type } = state.project;

  // ── Grundgrößen ──────────────────────────────────
  const S_mva    = (power / 1000) / cosphi;          // Scheinleistung MVA
  const U_kv     = voltage;
  const sqrt3    = Math.sqrt(3);
  const U_v      = U_kv * 1000;

  // Betriebsstrom
  const I_b      = (S_mva * 1e6) / (sqrt3 * U_v);   // A

  // ── Impedanzen ───────────────────────────────────
  // Netzimpedanz am VAP
  const Z_netz   = (U_kv * U_kv) / sk_vap;           // Ω (Betrag)
  const X_netz   = Z_netz / Math.sqrt(1 + rx*rx);
  const R_netz   = rx * X_netz;

  // Kabelimpedanz
  const R_kabel  = r_km * lengthKm / parallel;        // Ω
  const X_kabel  = x_km * lengthKm / parallel;        // Ω

  // Trafoimpedanz (auf MS-Seite)
  const Z_trafo  = uk * (U_kv*U_kv) / (trafoSn/1000); // Ω
  const X_trafo  = Z_trafo * 0.98;
  const R_trafo  = Z_trafo * 0.2;

  // Gesamt
  const R_ges    = R_netz + R_kabel + R_trafo;
  const X_ges    = X_netz + X_kabel + X_trafo;
  const Z_ges    = Math.sqrt(R_ges**2 + X_ges**2);

  // Kurzschlussleistung am Anschlusspunkt
  const sk_ap    = (U_kv * U_kv) / Z_ges;             // MVA

  // Kurzschlussstrom
  const ik_vap   = (sk_vap * 1e6) / (sqrt3 * U_v);   // kA (Betrag → in kA)

  // ── Sk-Verhältnis ────────────────────────────────
  // VDE-AR-N 4110: S_A / Sk'' ≤ 0,02 (2%) für schnelle Anlassen
  const sk_ratio = S_mva / sk_ap;                     // dimensionslos
  const sk_limit = 0.02;                               // 2%

  // ── Spannungsänderung Δu ─────────────────────────
  // Δu = (P·R + Q·X) / U²  (vereinfacht)
  const P_mw  = power / 1000;
  const Q_mvar= P_mw * Math.tan(Math.acos(cosphi));
  const du_abs= (P_mw * R_ges + Q_mvar * X_ges) / (U_kv * U_kv); // pu
  const du_pct= du_abs * 100;
  const du_limit = U_kv >= 110 ? 3.0 : 3.0;          // % (VDE-AR-N 4110 §6: 3%)

  // ── Thermische Kabelauslastung ────────────────────
  const I_total  = Math.sqrt(((power + existing)/1000 * 1e6 / (sqrt3 * U_v))**2);
  const cable_pct= (I_total / i_max) * 100;
  const cable_limit = 80;

  // ── Trafo-Auslastung ─────────────────────────────
  const trafo_load= ((power + existing) / 1000) / (trafoSn / 1000); // pu
  const trafo_pct = trafo_load * 100;
  const trafo_limit = 80;

  // ── N-1 Analyse ──────────────────────────────────
  // Bei N-1 fällt ein Kabelstrang aus → Strom verdoppelt sich (parallel=2) oder alles weg (parallel=1)
  const cable_n1_pct = parallel > 1 ? (I_total / (i_max_1 * (parallel-1))) * 100 : 999;
  const trafo_n1_pct = trafo_pct * 1.0; // Reservetrafo notwendig wenn > 100%

  // ── Bewertung ────────────────────────────────────
  const checks = [
    { label: `Sk-Verhältnis S/Sk'' = ${(sk_ratio*100).toFixed(2)}% (Limit: ${sk_limit*100}%)`,
      pass: sk_ratio <= sk_limit,
      warn: sk_ratio <= sk_limit * 1.5,
      detail: `S=${S_mva.toFixed(2)} MVA | Sk''=${sk_ap.toFixed(1)} MVA am AP` },
    { label: `Spannungsänderung Δu = ${du_pct.toFixed(2)}% (Limit: ${du_limit}%)`,
      pass: du_pct <= du_limit,
      warn: du_pct <= du_limit * 1.2,
      detail: `VDE-AR-N 4110 §6 / EN 50160` },
    { label: `Thermische Kabelauslastung = ${cable_pct.toFixed(1)}% (Limit: ${cable_limit}%)`,
      pass: cable_pct <= cable_limit,
      warn: cable_pct <= 100,
      detail: `Ib=${I_total.toFixed(0)} A | Imax=${i_max} A (${parallel}× Strang)` },
    { label: `Trafo-Auslastung = ${trafo_pct.toFixed(1)}% (Limit: ${trafo_limit}%)`,
      pass: trafo_pct <= trafo_limit,
      warn: trafo_pct <= 100,
      detail: `${((power+existing)/1000).toFixed(1)} MVA / ${(trafoSn/1000).toFixed(1)} MVA` },
  ];

  const failCount = checks.filter(c => !c.pass && !c.warn).length;
  const warnCount = checks.filter(c => !c.pass && c.warn).length;
  const ampel = failCount > 0 ? 'red' : warnCount > 0 ? 'yellow' : 'green';

  // ── Empfehlungen ─────────────────────────────────
  const recs = [];
  if (sk_ratio > sk_limit)   recs.push('🔁 Stärkere Netzanbindung / höherer Sk erforderlich. Netzbetreiber kontaktieren.');
  if (du_pct > du_limit)     recs.push('⚡ Spannungsänderung zu hoch → Blindleistungskompensation (SVC/STATCOM) oder kürzeres Kabel prüfen.');
  if (cable_pct > cable_limit) recs.push('🔌 Kabel überlastet → größeren Querschnitt (z.B. 240mm²) oder Parallelstränge einplanen.');
  if (trafo_pct > trafo_limit) recs.push('🏭 Trafo überlastet → größeren Transformator oder zweiten Trafo (N-1) vorsehen.');
  if (parallel === 1 && cable_pct > 40) recs.push('🔄 N-1: Bei Kabelausfall kein Redundanzpfad → zweiten Kabelstrang empfohlen.');
  if (recs.length === 0)     recs.push('✅ Alle Prüfkriterien erfüllt. Detailplanung und Netzverträglichkeitsstudie beim Netzbetreiber beantragen.');

  // ── Ergebnis-Objekt ──────────────────────────────
  const result = {
    ampel, checks, recs,
    metrics: {
      sk_ratio_pct: (sk_ratio*100).toFixed(2), sk_limit_pct: (sk_limit*100).toFixed(0),
      sk_ap: sk_ap.toFixed(1),
      du_pct: du_pct.toFixed(2), du_limit,
      cable_pct: cable_pct.toFixed(1), cable_limit,
      i_total: I_total.toFixed(0), i_max,
      trafo_pct: trafo_pct.toFixed(1), trafo_limit,
    },
    details: {
      S_mva: S_mva.toFixed(3),
      cosphi,
      sk_vap,
      sk_ap: sk_ap.toFixed(1),
      ik_vap: (ik_vap/1000).toFixed(2),
      Z_netz: Z_netz.toFixed(3),
      R_kabel: R_kabel.toFixed(3), X_kabel: X_kabel.toFixed(3),
      Z_trafo: Z_trafo.toFixed(3),
      R_ges: R_ges.toFixed(3), X_ges: X_ges.toFixed(3), Z_ges: Z_ges.toFixed(3),
      I_b: I_b.toFixed(1),
      cable_n1: cable_n1_pct > 900 ? 'Kein Redundanzpfad' : cable_n1_pct.toFixed(1)+'%',
      trafo_n1: trafo_n1_pct > 100 ? 'Reservetrafo nötig' : 'OK ('+trafo_n1_pct.toFixed(1)+'%)',
    },
    project: { name, plz, type, power, voltage: U_kv, cosphi },
    ts: new Date().toLocaleString('de-DE')
  };

  state.result = result;
  state.calcId = 'GC-' + Date.now().toString(36).toUpperCase();

  // Verlauf speichern
  history.unshift({ id: state.calcId, name, power, voltage: U_kv, ampel,
                    du: result.metrics.du_pct, cable: result.metrics.cable_pct, ts: result.ts });
  if (history.length > 20) history.pop();
  localStorage.setItem('gc_history', JSON.stringify(history));

  renderResult(result);
  goStep(3);
  renderHistory();
}

function col(val, limit, inverted) {
  const v = parseFloat(val);
  const l = parseFloat(limit);
  if (!inverted) {
    if (v <= l * 0.8)  return 'var(--green)';
    if (v <= l)        return 'var(--yellow)';
    return 'var(--red)';
  }
  return v >= l ? 'var(--green)' : 'var(--red)';
}

function barW(val, limit) {
  return Math.min((parseFloat(val) / parseFloat(limit)) * 100, 100).toFixed(0);
}

function renderHistory() {

  const colorMap = { green: 'var(--green)', yellow: 'var(--yellow)', red: 'var(--red)' };
  document.getElementById('history-list').innerHTML = history.map(h => `

      <div>
        <span style="color:${colorMap[h.ampel] || 'gray'}">●</span>
        <strong style="margin-left:6px">${h.name}</strong>
        <span style="color:var(--muted);font-size:12px"> | ${h.power.toLocaleString('de')} kW @ ${h.voltage} kV | Δu=${h.du}% | Kabel ${h.cable}%</span>
      </div>
      <span style="font-size:11px;color:var(--muted)">${h.id} | ${h.ts}</span>
    </div>
  `).join('');
}

// EXPORT
// ═══════════════════════════════════════════════════
function exportJSON() {
  if (!state.result) return;
  const blob = new Blob([JSON.stringify({ id: state.calcId, ...state.result }, null, 2)],
                        { type: 'application/json' });
  const a = document.createElement('a');
  a.href = URL.createObjectURL(blob);
  a.download = `GridCheck_${state.calcId}_${state.result.project.name.replace(/\s/g,'_')}.json`;
  a.click();
}

function exportPDF() {
  window.print();
}
