import { useState } from 'react';

const ANLAGEN_TYPEN = [
  { value: 'pv', label: '☀️ Photovoltaik' },
  { value: 'wind', label: '💨 Windenergie' },
  { value: 'bess', label: '🔋 Batteriespeicher' },
  { value: 'biogas', label: '🌿 Biogas' },
  { value: 'last', label: '🏭 Großverbraucher' },
];

const SPANNUNGSEBENEN = [
  { value: '0.4', label: '0,4 kV (NS)' },
  { value: '10', label: '10 kV (MS)' },
  { value: '20', label: '20 kV (MS)' },
  { value: '110', label: '110 kV (HS)' },
];

export default function StepProjekt({ data, onNext }) {
  const [form, setForm] = useState({
    projektname: data.projektname || '',
    plz: data.plz || '',
    anlagentyp: data.anlagentyp || 'pv',
    leistung_kw: data.leistung_kw || '',
    spannungsebene_kv: data.spannungsebene_kv || '20',
    cos_phi: data.cos_phi || '0,95',
  });

  const [errors, setErrors] = useState({});

  const validate = () => {
    const e = {};
    if (!form.projektname.trim()) e.projektname = 'Pflichtfeld';
    if (!form.plz.trim()) e.plz = 'Pflichtfeld';
    if (!form.leistung_kw || parseFloat(String(form.leistung_kw).replace(',', '.')) <= 0) {
      e.leistung_kw = 'Leistung > 0 erforderlich';
    }
    const cosVal = parseFloat(String(form.cos_phi).replace(',', '.'));
    if (isNaN(cosVal) || cosVal < 0.8 || cosVal > 1.0) {
      e.cos_phi = 'Wert zwischen 0,80 und 1,00';
    }
    setErrors(e);
    return Object.keys(e).length === 0;
  };

  const handleSubmit = () => {
    if (!validate()) return;
    onNext({
      projektname: form.projektname.trim(),
      plz: form.plz.trim(),
      anlagentyp: form.anlagentyp,
      leistung_kw: parseFloat(String(form.leistung_kw).replace(',', '.')),
      spannungsebene_kv: parseFloat(form.spannungsebene_kv),
      cos_phi: parseFloat(String(form.cos_phi).replace(',', '.')),
    });
  };

  const set = (key, val) => setForm({ ...form, [key]: val });

  const inputCls = (field) =>
    `w-full bg-gray-900 border ${errors[field] ? 'border-red-500' : 'border-gray-600'} rounded-lg px-4 py-3 text-white focus:outline-none focus:border-blue-500 transition`;

  return (
    <div>
      <h2 className="text-xl font-bold text-blue-400 mb-6">Projektdaten</h2>
      <div className="grid grid-cols-1 md:grid-cols-2 gap-5">
        <div>
          <label className="block text-xs font-semibold text-gray-400 mb-1 uppercase tracking-wide">Projektname *</label>
          <input className={inputCls('projektname')} value={form.projektname} onChange={e => set('projektname', e.target.value)} placeholder="z.B. Solarpark Süd" />
          {errors.projektname && <p className="text-red-400 text-xs mt-1">{errors.projektname}</p>}
        </div>
        <div>
          <label className="block text-xs font-semibold text-gray-400 mb-1 uppercase tracking-wide">PLZ *</label>
          <input className={inputCls('plz')} value={form.plz} onChange={e => set('plz', e.target.value)} placeholder="z.B. 80331" />
          {errors.plz && <p className="text-red-400 text-xs mt-1">{errors.plz}</p>}
        </div>
        <div>
          <label className="block text-xs font-semibold text-gray-400 mb-1 uppercase tracking-wide">Anlagentyp *</label>
          <select className={inputCls('anlagentyp')} value={form.anlagentyp} onChange={e => set('anlagentyp', e.target.value)}>
            {ANLAGEN_TYPEN.map(t => <option key={t.value} value={t.value}>{t.label}</option>)}
          </select>
        </div>
        <div>
          <label className="block text-xs font-semibold text-gray-400 mb-1 uppercase tracking-wide">Anschlussleistung (kW) *</label>
          <input className={inputCls('leistung_kw')} type="number" min="0" value={form.leistung_kw} onChange={e => set('leistung_kw', e.target.value)} placeholder="z.B. 5000" />
          {errors.leistung_kw && <p className="text-red-400 text-xs mt-1">{errors.leistung_kw}</p>}
        </div>
        <div>
          <label className="block text-xs font-semibold text-gray-400 mb-1 uppercase tracking-wide">Spannungsebene</label>
          <select className={inputCls('spannungsebene_kv')} value={form.spannungsebene_kv} onChange={e => set('spannungsebene_kv', e.target.value)}>
            {SPANNUNGSEBENEN.map(s => <option key={s.value} value={s.value}>{s.label}</option>)}
          </select>
        </div>
        <div>
          <label className="block text-xs font-semibold text-gray-400 mb-1 uppercase tracking-wide">cos φ</label>
          <input className={inputCls('cos_phi')} value={form.cos_phi} onChange={e => set('cos_phi', e.target.value)} />
          {errors.cos_phi && <p className="text-red-400 text-xs mt-1">{errors.cos_phi}</p>}
          <p className="text-gray-600 text-xs mt-1">Standard: 0,95 (VDE-AR-N 4110)</p>
        </div>
      </div>
      <div className="flex justify-end mt-8">
        <button onClick={handleSubmit} className="bg-blue-600 hover:bg-blue-500 px-8 py-3 rounded-lg font-bold transition">
          Weiter → Netzparameter
        </button>
      </div>
    </div>
  );
}
