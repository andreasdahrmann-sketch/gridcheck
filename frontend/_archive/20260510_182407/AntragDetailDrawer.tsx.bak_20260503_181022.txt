"use client";

import React, { useEffect, useState } from "react";
import { X, AlertTriangle, Sparkles, Network, Clock, FileText, Shield, ShieldAlert } from "lucide-react";
import type { Antrag, PrioResult } from "@/lib/priorisierung";
import { GEWICHTE } from "@/lib/priorisierung";
import {
  type AntragStatus,
  type AuditEntry,
  STATUS_LABEL,
  STATUS_COLOR,
  getMeta,
  getAuditTrail,
  setStatus,
  setNote,
  verifyAuditChain,
} from "@/lib/antragStore";

const TYPEN_LABEL: Record<string, string> = {
  solar: "PV",
  wind: "Wind",
  batterie: "Batteriespeicher",
  ladepark: "Ladepark",
  industrie: "Industrie",
  waermepumpe: "Waermepumpe",
  bhkw: "BHKW",
  sonstige: "Sonstige",
};

const REIFE_LABEL: Record<string, string> = {
  idee: "Idee",
  planung: "Planung",
  genehmigt: "Genehmigt",
  baubereit: "Baubereit",
};

const CURRENT_USER = "netzbetreiber@demo";

function scoreColor(score: number): string {
  if (score >= 75) return "text-emerald-400";
  if (score >= 50) return "text-amber-400";
  return "text-rose-400";
}

function scoreBg(score: number): string {
  if (score >= 75) return "bg-emerald-500";
  if (score >= 50) return "bg-amber-500";
  return "bg-rose-500";
}

function formatTs(iso: string): string {
  try {
    return new Date(iso).toLocaleString("de-DE", {
      day: "2-digit", month: "2-digit", year: "numeric",
      hour: "2-digit", minute: "2-digit",
    });
  } catch {
    return iso;
  }
}

interface Props {
  open: boolean;
  onClose: () => void;
  antrag: Antrag | null;
  prio: PrioResult | null;
}

export default function AntragDetailDrawer({ open, onClose, antrag, prio }: Props) {
  const [currentStatus, setCurrentStatus] = useState<AntragStatus>("neu");
  const [statusComment, setStatusComment] = useState("");
  const [noteInput, setNoteInput] = useState("");
  const [savedNote, setSavedNote] = useState("");
  const [auditTrail, setAuditTrail] = useState<AuditEntry[]>([]);
  const [chainValid, setChainValid] = useState<boolean>(true);
  const [feedback, setFeedback] = useState<string>("");

  useEffect(() => {
    if (!open || !antrag) return;
    const meta = getMeta(antrag.id);
    setCurrentStatus(meta.status);
    setSavedNote(meta.note || "");
    setNoteInput(meta.note || "");
    setStatusComment("");
    const trail = getAuditTrail(antrag.id);
    setAuditTrail(trail);
    const verify = verifyAuditChain(antrag.id);
    setChainValid(verify.valid);
    setFeedback("");
  }, [open, antrag]);

  if (!open || !antrag || !prio) return null;

  const handleStatusChange = (newStatus: AntragStatus) => {
    if (newStatus === currentStatus) return;
    setStatus(antrag.id, newStatus, statusComment, CURRENT_USER);
    setCurrentStatus(newStatus);
    setStatusComment("");
    setAuditTrail(getAuditTrail(antrag.id));
    setChainValid(verifyAuditChain(antrag.id).valid);
    setFeedback("Status aktualisiert.");
    setTimeout(() => setFeedback(""), 2000);
  };

  const handleNoteSave = () => {
    if (noteInput === savedNote) return;
    setNote(antrag.id, noteInput, CURRENT_USER);
    setSavedNote(noteInput);
    setAuditTrail(getAuditTrail(antrag.id));
    setChainValid(verifyAuditChain(antrag.id).valid);
    setFeedback("Notiz gespeichert.");
    setTimeout(() => setFeedback(""), 2000);
  };

  const statusOptions: AntragStatus[] = ["neu", "in_pruefung", "rueckfrage", "genehmigt", "abgelehnt", "abgeschlossen"];

  return (
    <div className="fixed inset-0 z-50 flex">
      <div className="flex-1 bg-black/60 backdrop-blur-sm" onClick={onClose} />
      <div className="w-full max-w-2xl h-full bg-slate-900 border-l border-slate-700 overflow-y-auto">
        <div className="sticky top-0 z-10 bg-slate-900 border-b border-slate-700 p-4 flex items-center justify-between">
          <div>
            <h2 className="text-xl font-bold text-white">{antrag.name}</h2>
            <p className="text-sm text-slate-400">ID: {antrag.id}</p>
          </div>
          <button onClick={onClose} className="p-2 hover:bg-slate-800 rounded text-slate-400 hover:text-white">
            <X className="w-5 h-5" />
          </button>
        </div>

        <div className="p-6 space-y-6">
          {feedback && (
            <div className="bg-emerald-500/10 border border-emerald-500/30 text-emerald-300 px-4 py-2 rounded text-sm">
              {feedback}
            </div>
          )}

          {!chainValid && (
            <div className="bg-rose-500/10 border border-rose-500/30 text-rose-300 px-4 py-3 rounded text-sm flex items-start gap-2">
              <ShieldAlert className="w-5 h-5 flex-shrink-0 mt-0.5" />
              <div>
                <div className="font-semibold">Audit-Chain ungueltig!</div>
                <div className="text-xs mt-1">Die Hash-Kette wurde manipuliert oder ist beschaedigt.</div>
              </div>
            </div>
          )}

          <section className="bg-slate-800/50 rounded-lg p-4 border border-slate-700">
            <div className="flex items-center gap-2 mb-3">
              <FileText className="w-4 h-4 text-cyan-400" />
              <h3 className="font-semibold text-white">Stammdaten</h3>
            </div>
            <div className="grid grid-cols-2 gap-3 text-sm">
              <div><span className="text-slate-400">Typ:</span> <span className="text-white">{TYPEN_LABEL[antrag.typ] || antrag.typ}</span></div>
              <div><span className="text-slate-400">Leistung:</span> <span className="text-white">{antrag.leistung_kw} kW</span></div>
              <div><span className="text-slate-400">Reife:</span> <span className="text-white">{REIFE_LABEL[antrag.reife] || antrag.reife}</span></div>
              <div><span className="text-slate-400">Eingang:</span> <span className="text-white">{formatTs(antrag.eingang)}</span></div>
              {antrag.netzknoten && <div><span className="text-slate-400">Netzknoten:</span> <span className="text-white">{antrag.netzknoten}</span></div>}
              {antrag.spannungsebene && <div><span className="text-slate-400">Spannungsebene:</span> <span className="text-white">{antrag.spannungsebene}</span></div>}
            </div>
          </section>

          <section className="bg-slate-800/50 rounded-lg p-4 border border-slate-700">
            <div className="flex items-center justify-between mb-3">
              <div className="flex items-center gap-2">
                <Sparkles className="w-4 h-4 text-cyan-400" />
                <h3 className="font-semibold text-white">Prioritaets-Score</h3>
              </div>
              <div className={`text-3xl font-bold ${scoreColor(prio.score)}`}>{prio.score}</div>
            </div>
            <div className="w-full bg-slate-700 rounded-full h-2 mb-4">
              <div className={`h-2 rounded-full ${scoreBg(prio.score)}`} style={{ width: `${prio.score}%` }} />
            </div>
            <div className="space-y-2 text-xs">
              {Object.entries(prio.komponenten).map(([key, val]) => (
                <div key={key} className="flex items-center justify-between">
                  <span className="text-slate-400 capitalize">{key} <span className="text-slate-500">(x{GEWICHTE[key as keyof typeof GEWICHTE]})</span></span>
                  <span className="text-white font-mono">{val}</span>
                </div>
              ))}
            </div>
            {prio.flags && prio.flags.length > 0 && (
              <div className="mt-3 pt-3 border-t border-slate-700 flex items-start gap-2">
                <AlertTriangle className="w-4 h-4 text-amber-400 flex-shrink-0 mt-0.5" />
                <div className="flex flex-wrap gap-1">
                  {prio.flags.map((f, i) => (
                    <span key={i} className="px-2 py-0.5 bg-amber-500/10 text-amber-300 text-xs rounded border border-amber-500/30">{f}</span>
                  ))}
                </div>
              </div>
            )}
          </section>

          <section className="bg-slate-800/50 rounded-lg p-4 border border-slate-700">
            <div className="flex items-center gap-2 mb-3">
              <Network className="w-4 h-4 text-cyan-400" />
              <h3 className="font-semibold text-white">Status</h3>
            </div>
            <div className="grid grid-cols-2 gap-2 mb-3">
              {statusOptions.map((s) => (
                <button
                  key={s}
                  onClick={() => handleStatusChange(s)}
                  className={`px-3 py-2 rounded text-sm font-medium transition border ${
                    currentStatus === s
                      ? `${STATUS_COLOR[s]} border-current`
                      : "bg-slate-900 text-slate-400 border-slate-700 hover:border-slate-500"
                  }`}
                >
                  {STATUS_LABEL[s]}
                </button>
              ))}
            </div>
            <input
              type="text"
              value={statusComment}
              onChange={(e) => setStatusComment(e.target.value)}
              placeholder="Optionaler Kommentar zur Statusaenderung..."
              className="w-full px-3 py-2 bg-slate-900 border border-slate-700 rounded text-sm text-white placeholder-slate-500 focus:border-cyan-500 outline-none"
            />
          </section>

          <section className="bg-slate-800/50 rounded-lg p-4 border border-slate-700">
            <div className="flex items-center gap-2 mb-3">
              <FileText className="w-4 h-4 text-cyan-400" />
              <h3 className="font-semibold text-white">Notiz</h3>
            </div>
            <textarea
              value={noteInput}
              onChange={(e) => setNoteInput(e.target.value)}
              rows={3}
              placeholder="Interne Notiz..."
              className="w-full px-3 py-2 bg-slate-900 border border-slate-700 rounded text-sm text-white placeholder-slate-500 focus:border-cyan-500 outline-none resize-none"
            />
            <button
              onClick={handleNoteSave}
              disabled={noteInput === savedNote}
              className="mt-2 px-4 py-2 bg-cyan-600 hover:bg-cyan-500 disabled:bg-slate-700 disabled:text-slate-500 text-white rounded text-sm font-medium transition"
            >
              Notiz speichern
            </button>
          </section>

          <section className="bg-slate-800/50 rounded-lg p-4 border border-slate-700">
            <div className="flex items-center justify-between mb-3">
              <div className="flex items-center gap-2">
                <Clock className="w-4 h-4 text-cyan-400" />
                <h3 className="font-semibold text-white">Audit-Trail</h3>
                <span className="text-xs text-slate-500">({auditTrail.length})</span>
              </div>
              {chainValid && (
                <div className="flex items-center gap-1 text-xs text-emerald-400">
                  <Shield className="w-3 h-3" />
                  <span>Chain gueltig</span>
                </div>
              )}
            </div>
            {auditTrail.length === 0 ? (
              <p className="text-sm text-slate-500 italic">Noch keine Eintraege.</p>
            ) : (
              <ol className="space-y-2 max-h-64 overflow-y-auto">
                {auditTrail.slice().reverse().map((e) => (
                  <li key={e.id} className="text-xs border-l-2 border-slate-700 pl-3 py-1">
                    <div className="flex items-center justify-between">
                      <span className="text-cyan-400 font-medium">{e.action}</span>
                      <span className="text-slate-500">{formatTs(e.ts)}</span>
                    </div>
                    <div className="text-slate-400 mt-0.5">
                      {e.from && e.to && <span>{e.from} -&gt; {e.to}</span>}
                      {e.comment && <span className="italic"> "{e.comment}"</span>}
                    </div>
                    <div className="text-slate-600 mt-0.5">von {e.user}</div>
                  </li>
                ))}
              </ol>
            )}
          </section>
        </div>
      </div>
    </div>
  );
}
